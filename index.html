// ================================================================ // KitTrack — Google Apps Script Backend // Paste this entire file into Extensions > Apps Script in your Sheet // Then: Deploy > New deployment > Web app > Execute as: Me > Anyone // ================================================================ const SHEET_ID = SpreadsheetApp.getActiveSpreadsheet().getId(); // Sheet tab names — these will be auto-created if they don't exist const TABS = { kits: "Kits", reqs: "Requests", txns: "Transactions", reports: "Reports" }; // ── Entry point for all requests ────────────────────────────────── function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const action = payload.action; let result; if (action === "load") result = loadAll(); else if (action === "saveKits") result = saveKits(payload.kits); else if (action === "addReq") result = addRow(TABS.reqs, reqToRow(payload.req)); else if (action === "updateReq") result = updateReq(payload.id, payload.fields); else if (action === "deleteReq") result = deleteRow(TABS.reqs, payload.id); else if (action === "addTxn") result = addRow(TABS.txns, txnToRow(payload.txn)); else if (action === "addReport") result = addRow(TABS.reports, rptToRow(payload.report)); else if (action === "deleteReport") result = deleteRow(TABS.reports, payload.id); else result = { error: "Unknown action: " + action }; return respond(result); } catch(err) { return respond({ error: err.toString() }); } } function doGet(e) { // Allow GET for load (easier for initial fetch) return respond(loadAll()); } // ── Load all data ───────────────────────────────────────────────── function loadAll() { ensureSheets(); return { kits: getRows(TABS.kits).map(rowToKit), reqs: getRows(TABS.reqs).map(rowToReq), txns: getRows(TABS.txns).map(rowToTxn), reports: getRows(TABS.reports).map(rowToRpt) }; } // ── Kit helpers ─────────────────────────────────────────────────── function saveKits(kits) { const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(TABS.kits); // Clear data rows (keep header) if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow()-1, sh.getLastColumn()).clearContent(); kits.forEach(k => sh.appendRow([k.id, k.name, k.status, k.checkedOutTo||"", k.checkedOutSince||"", k.event||"", k.lastVerified||""])); return { ok: true }; } function rowToKit(r) { return { id:r[0], name:r[1], status:r[2], checkedOutTo:r[3]||null, checkedOutSince:r[4]||null, event:r[5]||null, lastVerified:r[6]||"" }; } // ── Request helpers ─────────────────────────────────────────────── function reqToRow(r) { return [r.id, r.name, r.email||"", r.department||"", r.program, r.event, r.location||"", r.dateNeeded, r.startTime||"", r.dateReturn||"", r.retTime||"", r.population||"", r.notes||"", r.status, r.assignedKit||"", r.submittedAt]; } function rowToReq(r) { return { id:r[0], name:r[1], email:r[2], department:r[3], program:r[4], event:r[5], location:r[6], dateNeeded:r[7], startTime:r[8], dateReturn:r[9], retTime:r[10], population:r[11], notes:r[12], status:r[13], assignedKit:r[14]||null, submittedAt:r[15] }; } function updateReq(id, fields) { const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(TABS.reqs); const data = sh.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === id) { // status is col 14 (index 13), assignedKit is col 15 (index 14) if (fields.status !== undefined) sh.getRange(i+1, 14).setValue(fields.status); if (fields.assignedKit !== undefined) sh.getRange(i+1, 15).setValue(fields.assignedKit||""); return { ok: true }; } } return { error: "Row not found: " + id }; } // ── Transaction helpers ─────────────────────────────────────────── function txnToRow(t) { return [t.id, t.type, t.kit, t.person, t.event||"", t.timestamp, t.notes||""]; } function rowToTxn(r) { return { id:r[0], type:r[1], kit:r[2], person:r[3], event:r[4], timestamp:r[5], notes:r[6] }; } // ── Report helpers ──────────────────────────────────────────────── function rptToRow(r) { return [r.id, r.name, r.event, r.attendance, r.population||"", r.kitCondition, r.went, r.improve||"", r.submittedAt]; } function rowToRpt(r) { return { id:r[0], name:r[1], event:r[2], attendance:r[3], population:r[4], kitCondition:r[5], went:r[6], improve:r[7], submittedAt:r[8] }; } // ── Generic sheet helpers ───────────────────────────────────────── function getRows(tabName) { const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(tabName); if (!sh || sh.getLastRow() < 2) return []; return sh.getRange(2, 1, sh.getLastRow()-1, sh.getLastColumn()).getValues() .filter(r => r[0] !== ""); // skip blank rows } function addRow(tabName, row) { const ss = SpreadsheetApp.openById(SHEET_ID); ss.getSheetByName(tabName).appendRow(row); return { ok: true }; } function deleteRow(tabName, id) { const ss = SpreadsheetApp.openById(SHEET_ID); const sh = ss.getSheetByName(tabName); const data = sh.getDataRange().getValues(); for (let i = data.length - 1; i >= 1; i--) { if (data[i][0] === id) { sh.deleteRow(i+1); return { ok: true }; } } return { error: "Row not found: " + id }; } // ── Ensure all sheets + headers exist ──────────────────────────── function ensureSheets() { const ss = SpreadsheetApp.openById(SHEET_ID); const headers = { [TABS.kits]: ["id","name","status","checkedOutTo","checkedOutSince","event","lastVerified"], [TABS.reqs]: ["id","name","email","department","program","event","location","dateNeeded","startTime","dateReturn","retTime","population","notes","status","assignedKit","submittedAt"], [TABS.txns]: ["id","type","kit","person","event","timestamp","notes"], [TABS.reports]: ["id","name","event","attendance","population","kitCondition","went","improve","submittedAt"] }; Object.entries(headers).forEach(([name, cols]) => { let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(cols); sh.getRange(1,1,1,cols.length).setFontWeight("bold").setBackground("#f3f4f6"); sh.setFrozenRows(1); } }); // Seed kits if Kits tab is empty const kitsSh = ss.getSheetByName(TABS.kits); if (kitsSh.getLastRow() < 2) { kitsSh.appendRow(["kit-a","Outreach Kit A","available","","","","2026-02-28"]); kitsSh.appendRow(["kit-b","Outreach Kit B","available","","","","2026-03-01"]); } } // ── Response helper ─────────────────────────────────────────────── function respond(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
